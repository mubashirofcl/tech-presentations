Section 1: Implementing CRUD API

CREATE — Adding a New Student

Implementing POST /students with body validation and HTTP 201 Created

Slide 05 / 18
server.js (POST /students)
app.post("/students", (req, res) => {
    const { name, email, course } = req.body;

    // 1. Validate required fields
    if (!name || !email || !course) {
        return res.status(400).json({
            message: "All fields are required"
        });
    }

    // 2. Auto-increment dynamic ID
    const newStudent = {
        id: students.length
            ? Math.max(...students.map(s => s.id)) + 1
            : 1,
        name,
        email,
        course
    };

    // 3. Save to array and respond with 201 Created
    students.push(newStudent);

    res.status(201).json({
        message: "Student created successfully",
        data: newStudent
    });
});
POST Testing CREATE in Postman

  • Body Tab: Select raw → JSON
Postman Request Body & 201 Response
// JSON REQUEST BODY: { "name": "Mubashir", "email": "mubashir@gmail.com", "course": "MERN Stack" } // STATUS: 201 Created { "message": "Student created successfully", "data": { "id": 3, "name": "Mubashir", "email": "mubashir@gmail.com", "course": "MERN Stack" } }
Key Point: 201 Created explicitly signals that a new resource was successfully added to our data collection.

UPDATE — Updating Student Information

Replacing a complete student record using the PUT /students/:id method

Slide 07 / 18
server.js (PUT /students/:id)
app.put("/students/:id", (req, res) => {
    const id = Number(req.params.id);

    // 1. Locate existing student
    const student = students.find(s => s.id === id);
    if (!student) {
        return res.status(404).json({
            message: "Student not found"
        });
    }

    const { name, email, course } = req.body;

    // 2. PUT requires all representation fields
    if (!name || !email || !course) {
        return res.status(400).json({
            message: "All fields are required"
        });
    }

    // 3. Update the fields
    student.name = name;
    student.email = email;
    student.course = course;

    res.status(200).json({
        message: "Student updated successfully",
        data: student
    });
});
PUT The Nature of HTTP PUT

PUT replaces the complete representation of a resource. The client must submit all required fields (name, email, and course).

Postman: PUT /students/1
// REQUEST BODY: { "name": "Anu Updated", "email": "anuupdated@gmail.com", "course": "Node.js" } // STATUS: 200 OK { "message": "Student updated successfully", "data": { "id": 1, "name": "Anu Updated", "email": "anuupdated@gmail.com", "course": "Node.js" } }
Section 1: Implementing CRUD API

PATCH — Updating Selected Fields

Modifying only specified fields with PATCH /students/:id and Object.assign()

Slide 08 / 18
server.js (PATCH /students/:id)
app.patch("/students/:id", (req, res) => {
    const id = Number(req.params.id);

    const student = students.find(s => s.id === id);
    if (!student) {
        return res.status(404).json({ message: "Student not found" });
    }

    // 1. Allowed fields whitelist
    const allowedFields = ["name", "email", "course"];
    const updates = Object.keys(req.body);

    // 2. Check for empty or invalid fields
    if (updates.length === 0 || updates.some(f => !allowedFields.includes(f))) {
        return res.status(400).json({
            message: "Provide valid fields to update"
        });
    }

    // 3. Merge only provided fields into student
    Object.assign(student, req.body);

    res.status(200).json({
        message: "Student partially updated",
        data: student
    });
});
PATCH PUT vs PATCH Summary
  • PUT: Replaces the entire resource. All fields are mandatory.
  • PATCH: Updates only specific fields. Unmentioned properties remain unchanged!
Postman: PATCH /students/1
// REQUEST BODY (Only course is sent): { "course": "Advanced MERN" } // STATUS: 200 OK { "message": "Student partially updated", "data": { "id": 1, "name": "Anu", "email": "anu@gmail.com", "course": "Advanced MERN" } }

DELETE — Removing a Student

Finding record index, splicing from array, and confirming deletion

Slide 09 / 18
server.js (DELETE /students/:id)
app.delete("/students/:id", (req, res) => {
    const id = Number(req.params.id);

    // 1. Locate array index
    const studentIndex = students.findIndex(
        s => s.id === id
    );

    // 2. If student does not exist, return 404
    if (studentIndex === -1) {
        return res.status(404).json({
            message: "Student not found"
        });
    }

    // 3. Remove 1 item from array at index
    const deletedStudent = students.splice(studentIndex, 1)[0];

    res.status(200).json({
        message: "Student deleted successfully",
        data: deletedStudent
    });
});
DELETE Understanding Array Methods
  • findIndex() → Returns the array position (0, 1...) or -1 if not found.
  • splice(index, 1) → Mutates the array to remove the item and returns the removed record.
Postman: DELETE /students/2
// STATUS: 200 OK { "message": "Student deleted successfully", "data": { "id": 2, "name": "Rahul", "email": "rahul@gmail.com", "course": "Python" } }

            Section 2: Express Middleware
            
What Is Express Middleware?

            
Functions that execute in the middle between receiving a request and sending a response

          

          Slide 11 / 18
        


        

          
          

            

              

                🛡️
                Simple Definition
              

              

                Middleware is a JavaScript function that executes between receiving an HTTP request and sending an HTTP response.
              

              

                Think of middleware as a checkpoint or security guard. Before a request reaches your controller, middleware can inspect, validate, or modify it.
              

            


            

              

                🏢
                Real-World Office Analogy
              

              
Imagine visiting an office building:

              
    
                
  • 1. Security Gate: Security guards check your ID (Authentication).
    • 
                
  • 2. Reception Desk: Your visitor details are logged (Logging).
    • 
                
  • 3. Access Granted: You are allowed into the office room (Controller)!
    • 
              

            

          


          
          

            

              

                🔄
                The Middleware Pipeline
              

              

                Client Request

                    ↓

                Middleware Checkpoint (req, res, next)

                    ↓

                Route Handler (/students)

                    ↓

                Controller Function

                    ↓

                Response sent back to client
              

            


            

              What Can Middleware Do?
              

                
• Log incoming requests

                
• Parse request bodies

                
• Validate form data

                
• Verify login tokens

              

            

          

        

      


      
      

        

          

            Section 2: Express Middleware
            
Creating Custom Middleware

            
Writing a request logger, registering with app.use(), and understanding next()

          

          Slide 12 / 18
        


        

          
          

            

              

                
logger-demo.js

                
              

              
const express = require("express");
const app = express();

// 1. Custom Logger Middleware Function
const logger = (req, res, next) => {
    console.log(`${req.method} ${req.url}`);
    
    // CRITICAL: Hand over control to the next function
    next();
};

// 2. Register middleware globally across all routes
app.use(logger);

app.get("/", (req, res) => res.send("Home Page"));
app.get("/students", (req, res) => res.send("Student List"));

app.listen(3000);

            

          


          
          

            

              

                🔄
                The Crucial Role of next()
              

              

                next() is a callback provided by Express. It tells Express: "This checkpoint is finished, pass control to the next middleware or route handler!"
              

            


            

              
Console Output as Requests Arrive

              
GET /
GET /students
POST /students

            


            

              ⚠️
              

                The Hanging Request Trap: If your middleware does not send a response (res.send) and forgets to call next(), the browser will spin endlessly waiting for a response!
              

            

          

        

      


      
      

        

          

            Section 2: Express Middleware
            
Types of Express Middleware

            
Categorizing middleware by scope, origin, and responsibility

          

          Slide 13 / 18
        


        

          
          

            

              

                
                  

                    Middleware Type
                    Purpose & Scope
                  

                
                
                  

                    Application-level
                    Runs globally across every route via app.use()
                  

                  

                    Router-level
                    Binds to a specific mini-app instance via router.use()
                  

                  

                    Built-in
                    Shipped natively with Express (e.g. express.json())
                  

                  

                    Third-party
                    Installed from npm (e.g. cors, multer, morgan)
                  

                  

                    Error-handling
                    Catches errors using 4 parameters: (err, req, res, next)
                  

                
              

            


            

              💡
              

                Why Categorize? Different middleware solve different problems. Using them appropriately prevents repeating identical validation or logging across dozens of routes.
              

            

          


          
          

            

              

                
middleware-types.js

                
              

              
// 1. Application-level
app.use((req, res, next) => { console.log("Global app middleware"); next(); });

// 2. Router-level
const router = express.Router();
router.use((req, res, next) => { console.log("Student router only"); next(); });

// 3. Built-in
app.use(express.json());

// 4. Third-party
const cors = require("cors");
app.use(cors());

// 5. Error-handling (4 parameters)
app.use((err, req, res, next) => {
    res.status(500).json({ message: err.message });
});

            

          

        

      


      
      

        

          

            Section 2: Express Middleware
            
Implementing Validation Middleware

            
Creating reusable route validation to keep controllers clean and focused

          

          Slide 14 / 18
        


        

          
          

            

              

                
validateStudent.js

                
              

              
// 1. Reusable Validation Checkpoint
const validateStudent = (req, res, next) => {
    const { name, email, course } = req.body;

    if (!name || !email || !course) {
        return res.status(400).json({
            message: "All fields are required"
        });
    }

    // If valid, proceed to route handler
    next();
};

// 2. Attach validateStudent between route path & handler
app.post("/students", validateStudent, (req, res) => {
    // Guaranteed clean data reaches here!
    const newStudent = { id: students.length + 1, ...req.body };
    students.push(newStudent);
    res.status(201).json({ message: "Student created", data: newStudent });
});

            Section 2: Express Middleware
            
Error-Handling Middleware

            
Catching and centralizing application errors with the 4-parameter error handler

          

          Slide 15 / 18
        


        

          
          

            

              

                
error-handler-demo.js

                
              

              
// Route that throws or forwards an error
app.get("/error-test", (req, res, next) => {
    const error = new Error("Database connection failed");
    
    // Pass error to Express error pipeline
    next(error);
});

// Centralized Error-Handling Middleware (4 parameters!)
app.use((err, req, res, next) => {
    console.error("Error caught centrally:", err.message);

    res.status(500).json({
        success: false,
        message: err.message
    });
});

            

          


          
          

            

              

                ⚠️
                The 4-Parameter Signature
              

              

                Express identifies error handlers strictly by checking their parameter count:
              

              

                (err, req, res, next)
              

              

                You must write all four parameters, even if you don't use next inside the body.
              

            


            

              
Client JSON Response

              
// STATUS: 500 Internal Server Error
{
    "success": false,
    "message": "Database connection failed"
}

            


            

              📌
              

                Registration Order: Always register your error-handling middleware at the very bottom of server.js, after all other routes!
              

            

          

        

      


      
      

        

          

            Section 3: MVC Architecture
            
What Is MVC Architecture?

            
Dividing backend applications into Models, Views, and Controllers

          

          Slide 16 / 18
        


        

          
          

            

              

                🏛️
                The Three Responsibilities
              

              
    
                
  • 
                  1. Model (Data): Manages data structures, array operations, validations, and database queries.
                
    • 
                
  • 
                  2. View (Presentation): Renders visual user interfaces or HTML pages (React, templates).
                
    • 
                
  • 
                  3. Controller (Logic): Receives requests, calls the Model for data, and returns responses.
                
    • 
              

            


            

              💡
              

                Why Express REST APIs don't need a View folder: Modern Express backends send raw JSON directly from Controllers to clients (like React or Mobile apps). The frontend renders the view!
              

            

          


          
          

            

              

                🔄
                REST API MVC Flow
              

              

                Client Request (Postman / React)

                    ↓

                Controller (Receives req & prepares res)

                    ↓

                Model (Performs find, create, delete)

                    ↓

                Student Data Store (Array / DB)

                    ↓

                Controller returns JSON back to Client!

            Section 3: MVC Architecture
            
Connecting Model, Controller & Routes

            
Inspect the complete working implementation across all separate files

// models/studentModel.js
let students = [
    { id: 1, name: "Anu", email: "anu@gmail.com", course: "MERN" },
    { id: 2, name: "Rahul", email: "rahul@gmail.com", course: "Python" }
];

const getAllStudents = () => students;

const getStudentById = (id) => students.find(s => s.id === id);

const createStudent = (data) => {
    const newStudent = {
        id: students.length ? Math.max(...students.map(s => s.id)) + 1 : 1,
        ...data
    };
    students.push(newStudent);
    return newStudent;
};

const deleteStudent = (id) => {
    const index = students.findIndex(s => s.id === id);
    if (index === -1) return null;
    return students.splice(index, 1)[0];
};

module.exports = { getAllStudents, getStudentById, createStudent, deleteStudent };

              


              
              

                
// controllers/studentController.js
const Student = require("../models/studentModel");

const getStudents = (req, res) => {
    res.status(200).json({ data: Student.getAllStudents() });
};

const getStudent = (req, res) => {
    const student = Student.getStudentById(Number(req.params.id));
    if (!student) return res.status(404).json({ message: "Student not found" });
    res.status(200).json({ data: student });
};

const createStudent = (req, res) => {
    const student = Student.createStudent(req.body);
    res.status(201).json({ message: "Student created", data: student });
};

module.exports = { getStudents, getStudent, createStudent };

              


              
              

                
// routes/studentRoutes.js
const express = require("express");
const router = express.Router();
const studentController = require("../controllers/studentController");
const validateStudent = require("../middleware/validateStudent");

router.get("/", studentController.getStudents);
router.get("/:id", studentController.getStudent);
router.post("/", validateStudent, studentController.createStudent);

module.exports = router;

              


              
              

                
// server.js
const express = require("express");
const studentRoutes = require("./routes/studentRoutes");
const logger = require("./middleware/logger");

const app = express();
app.use(express.json());
app.use(logger);

// Mount router at /students
app.use("/students", studentRoutes);

app.listen(3000, () => console.log("Server running on port 3000"));
